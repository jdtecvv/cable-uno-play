    ssl_dhparam /etc/letsencrypt/ssl-dhparams.pem;

    # Redireccionar raíz a XUI login
    location = / {
        return 302 https://app.teleunotv.cr/5pUNs3U2/login;
    }

    # Proxy a XUI (todo lo demás)
    location / {
        proxy_pass http://127.0.0.1:81;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        
        # Reescribir redirects de XUI
        proxy_redirect http://app.teleunotv.cr:81/ https://app.teleunotv.cr/;
        proxy_redirect http://127.0.0.1:81/ https://app.teleunotv.cr/;
        
        # Para streaming sin buffer
        proxy_buffering off;
        proxy_read_timeout 300s;

    }
